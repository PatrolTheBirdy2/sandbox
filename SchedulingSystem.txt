import json
import os


def clear_screen():
    os.system("cls" if os.name == "nt" else "clear")


class Student:

    def __init__(self, name, schedule=None):
        self.student_name = name
        self.student_schedule = schedule if schedule is not None else []

    def add_course(self):
        period = str(input("Which period? "))
        for item in self.student_schedule:
            if str(item[0]) == period:
                print(f"Sorry, period {period} is occupied!")
                return

        start_time = input("Start time (e.g., 8:00 AM)? ")
        end_time = input("End time (e.g., 9:00 AM)? ")
        course_name = input("Which course? ")

        self.student_schedule.append(
            (period, start_time, end_time, course_name)
        )
        print("Course added successfully!")

    def edit_time_block(self):
        period_to_edit = str(
            input("Which period would you like to edit? ")
        )

        for i in range(len(self.student_schedule)):
            if str(self.student_schedule[i][0]) == period_to_edit:
                course_name = self.student_schedule[i][3]
                print(
                    f"Editing Period {period_to_edit} ({course_name})"
                )

                new_start = input("Enter new start time: ")
                new_end = input("Enter new end time: ")

                self.student_schedule[i] = (
                    period_to_edit,
                    new_start,
                    new_end,
                    course_name,
                )
                print("Time block updated!")
                return

        print("Period not found in schedule.")

    def remove_course(self):
        period_remove = str(
            input("What period would you like to remove? ")
        )
        for item in self.student_schedule:
            if str(item[0]) == period_remove:
                self.student_schedule.remove(item)
                print(f"Period {period_remove} removed!")
                return
        print("That period is not in your schedule.")

    def to_dict(self):
        return {
            "name": self.student_name,
            "schedule": self.student_schedule,
        }

def save_data(students_list, teachers_list):
    data = {
        "students": [s.to_dict() for s in students_list],
        "teachers": [t.to_dict() for t in teachers_list],
    }
    with open("school_data.json", "w") as file:
        json.dump(data, file, indent=4)
    print("\n[Progress saved to school_data.json!]")


def load_data():
    if not os.path.exists("school_data.json"):
        return [], []

    try:
        with open("school_data.json", "r") as file:
            data = json.load(file)
            students = [
                Student(s["name"], s["schedule"])
                for s in data.get("students", [])
            ]
            teachers = [
                Teacher(t["name"], t["schedule"])
                for t in data.get("teachers", [])
            ]
            return students, teachers
    except (json.JSONDecodeError, KeyError):
        return [], []
